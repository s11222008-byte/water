  <script>
    const intro = document.querySelector('#intro');
    const arTask = document.querySelector('#arTask');
    const quiz = document.querySelector('#quiz');

    const progress = document.querySelector('#progress');
    const toQuiz = document.querySelector('#toQuiz');

    const feedback = document.querySelector('#feedback');
    const conclusion = document.querySelector('#conclusion');

    function showStep(step) {
      intro.hidden = step !== 1;
      arTask.hidden = step !== 2;
      quiz.hidden = step !== 3;

      const titles = {
        1: '認識任務',
        2: '操作 AR',
        3: '完成觀察題'
      };

      progress.textContent =
        `步驟 ${step}／3：${titles[step]}`;

      window.scrollTo({
        top: 0,
        behavior: 'smooth'
      });
    }

    document.querySelector('#toAR')
      .addEventListener('click', () => {
        showStep(2);
      });

    document.querySelector('#openAR')
      .addEventListener('click', () => {
        // 僅確認學生點過 AR 連結。
        // GAS 無法據此確認學生實際操作了哪些水位。
        toQuiz.disabled = false;

        document.querySelector('#arReminder')
          .textContent =
            '完成 AR 觀察後，切換回這個分頁，' +
            '再點「我已完成觀察，開始作答」。';
      });

    toQuiz.addEventListener('click', () => {
      showStep(3);
    });

    document.querySelector('#backToAR')
      .addEventListener('click', () => {
        showStep(2);
      });

    document.querySelectorAll(
      'input[name="answer"]'
    ).forEach(input => {
      input.addEventListener('change', () => {
        feedback.hidden = true;
        conclusion.hidden = true;
      });
    });

    document.querySelector('#quizForm')
      .addEventListener('submit', event => {
        event.preventDefault();

        const selected = document.querySelector(
          'input[name="answer"]:checked'
        );

        if (!selected) {
          return;
        }

        feedback.hidden = false;

        if (selected.value === 'B') {
          feedback.className = 'correct';

          feedback.textContent =
            '答對了！浮萍隨著水面一起升降，' +
            '根沒有固定在水槽底部。';

          conclusion.hidden = false;

        } else {
          feedback.className = 'retry';

          feedback.textContent =
            '再觀察一次：先把水位調低，再調高。' +
            '浮萍是留在原高度，還是一直跟著水面移動？';

          conclusion.hidden = true;
        }
      });
  </script>
